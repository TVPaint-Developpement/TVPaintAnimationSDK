How to open a file requester?

 

This section explains how to open the TVPaint (or Windows©)  file requester window and save parameters in an ASCII file :  

static PIFileReq FileReq;		/* File requester structure */ 

/*******************************************************/ 

static void GereSave(PIFilter * PiFilter, char *name)		/* Saving function */
{
char temp[4096];
void *file;

	file=TVOpenStringFile(PiFilter, name, 1);		/* Creates (or erase) a configuration file (flag set to 1) */
	if(file==NULL)
	{
		TVWarning(PiFilter, "Can't open file !");
		return;
	} 
}

static void GereLoad(PIFilter * PiFilter,char *name)		/* Loading function */
{
char temp[4096];
void *file;

	file=TVOpenStringFile(PiFilter, name, 0);		/* Reads an existing configuration file (flag set to 0) */
	if(file==NULL)
	{
		TVWarning(PiFilter, "Can't open file !");
		return;
	}
}

int FAR PASCAL PI_Open(PIFilter * PiFilter)
{ 
	.....

	memset(&FileReq, 0, sizeof(PIFileReq));			/* Initializations of the FileReq structure parameters */
	FileReq.Icon=0;
	FileReq.Title="Load";
	strcpy(FileReq.Pattern, "*.*");
	strcpy(FileReq.Path, "");

	if(!TVAllocFileReq(PiFilter, &FileReq))			/* If the FileReq allocation fails ... */
	{
		TVWarning(PiFilter, "PIAllocFileReq Fail ! ");
		free(PiFilter->UserParametersPtr);
		PiFilter->UserParametersPtr=NULL;
		PiFilter->UserParametersSize=0;
		return 0;
	}

	....
	
	return 1;
}
